USE CtaLineaDb
GO

CREATE TABLE import.Associate_Translate
(
	Id			int IDENTITY (1, 1),
	A_Descr		varchar(200) NULL,
	Oly_Descr	varchar(200) NULL
);

INSERT INTO import.Associate_Translate
	(A_Descr, Oly_Descr)
	SELECT a.Description,
			i.DITTA
		FROM Dbo.Associates a
		LEFT JOIN  import.Oly i
			ON a.Description LIKE '%' + i.DITTA + '%'

GO

CREATE TABLE import.Car_Translate
(
	ID			int IDENTITY (1,1) PRIMARY KEY,
	CarId		uniqueidentifier NULL,
	RegNum		varchar(20) NULL,
	Oly_RegNum	varchar(50) NULL
);

INSERT INTO import.Car_Translate
	(CarId, RegNum)
SELECT  c.CarId,
		REPLACE(REPLACE(c.RegNumber, ' ', ''), '-', '') AS regNum
	FROM dbo.Cars  c;

WITH CTE_Oly_Cars
AS
(
	(
		SELECT REPLACE(REPLACE(i.MEZZO_TITOLARE, ' ', ''), '-', '') AS Oly_RegNum
			FROM import.Oly i
			WHERE i.MEZZO_TITOLARE IS NOT NULL
	) UNION (
		SELECT REPLACE(REPLACE(i.MEZZO_DI_RISERVA_1, ' ', ''), '-', '') AS Oly_RegNum
			FROM import.Oly i
			WHERE i.MEZZO_DI_RISERVA_1 IS NOT NULL
	) UNION (
		SELECT REPLACE(REPLACE(i.MEZZO_DI_RISERVA_2, ' ', ''), '-', '') AS Oly_RegNum
			FROM import.Oly i
			WHERE i.MEZZO_DI_RISERVA_2 IS NOT NULL
	)
)
-- SELECT * FROM CTE_Oly_Cars
UPDATE import.Car_Translate
	SET Oly_RegNum = c.Oly_RegNum
	FROM import.Car_Translate i
	INNER JOIN CTE_Oly_Cars c
		ON i.RegNum = c.Oly_RegNum;
GO

BEGIN TRAN;

DELETE FROM dbo.runs;

-- corse
INSERT INTO [dbo].[Runs]
			(RunId
           ,[ContractId]
           ,[Extra]
           ,[ContractRowNumber]
           ,[StartDate]
           ,[EndDate]
           ,[RequestedDays]
           ,[Note])
	SELECT 
			i.OlyId,
			1 AS ContractID,
			CASE WHEN i.n_righe_capitolato_928 LIKE '%extra%' THEN  1 ELSE 0 END AS Extra_Appalto,
			n_righe_capitolato_928,
	        [data_inizio],
			[data_fine],
			[nr_gg],
			[note_appalto]
		FROM import.Oly i;

-- Varianti delle corse
INSERT INTO [dbo].[RunVariations]
           ([RunId]
           ,[StartDate]
           ,[LineNumber]
           ,[RunNumber]
		   , [Path]
           ,[StartTime]
           ,[EndTime]
           ,[CalendarId]
           ,[RequestedFrequency]
		   , RequestedCapacity
           ,[Km]
           ,[Note])
	SELECT 
			i.OlyId,
			NULL AS startDate,
			CAST([Linea] AS int) AS Linea,
			CAST( (CASE WHEN [n_corsa] LIKE '%/%' THEN SUBSTRING([n_corsa], 1, CHARINDEX('/', [n_corsa])-1 ) ELSE [n_corsa] END) AS int) AS Corsa_1,
			ISNULL([percorso], '-- NESSUN PERCORSO --'),
			[ora_part],
			[ora_arr],
			1,
			[frequenza],
			ISNULL(TRY_CAST([Posti_autobus] AS int) , 0) AS posti,
			[Km_corsa_servizio],
			CASE 
			WHEN TRY_CAST ([n_corsa] AS int) IS NULL  
				AND TRY_CAST([Posti_autobus] AS INT) IS NULL 
				THEN 'nr corsa con problemi :' + [n_corsa]
					+ 'nr posti con dei problemi: ' + [Posti_autobus]
			WHEN TRY_CAST ([n_corsa] AS int) IS NULL  
				THEN 'nr corsa con problemi :' + [n_corsa]
			WHEN  TRY_CAST([Posti_autobus] AS INT) IS NULL
				THEN 'nr posti con dei problemi: ' + [Posti_autobus]
			ELSE NULL
			END AS note
		FROM import.Oly i;

-- nodi
WITH CTE_Nodes AS
(
SELECT * FROM 
(
(
	SELECT i.OlyId,
			1 AS Progr,
			i.ora_part AS Time,
			i.Cod_Fermata_Partenza AS CodFermata
		FROM import.Oly i
) UNION ALL (
	SELECT i.OlyId,
			2 AS Progr,
			i.ora_arr AS Time,
			i.Cod_Fermata_Arrivo AS CodFermata
		FROM import.Oly i
) 
) AS t
)
INSERT INTO dbo.RunNodes
		(RunVariationId, ProgrNumber, Hour, CollectionPointId)
	SELECT v.RunVariationId,
			n.Progr,
			n.Time,
			n.CodFermata
		FROM CTE_Nodes n
		INNER JOIN dbo.RunVariations v
			ON n.OlyId = v.RunId
			AND v.StartDate IS NULL
		LEFT JOIN dbo.CollectionPoints cp
			ON n.CodFermata = cp.CollectionPointId
		WHERE cp.CollectionPointId IS NOT NULL
		;

-- periodi
INSERT INTO dbo.RunPeriods
	(RunId)

		FROM import.Oly i;
