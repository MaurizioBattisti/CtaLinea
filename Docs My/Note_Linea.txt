    protected override async Task OnInitializedAsync()
    {
        this._servicesQueryDef.EndPoint = Constants.Endpoint_Cars;
        await Task.CompletedTask;
        this.StateHasChanged();
    }

    private async Task LoadData (LoadDataArgs args)
    {
        this.isLoading = true;
        this.StateHasChanged();

        string? baseFilter = null;
        if (this.AssociateId != null)
        {
            baseFilter = string.Format("$eq(associateId,'{0}'", AssociateId.Value);
        }
        this._servicesQueryDef.Filter = args.GetQueryFilter<Car> (baseFilter);
        this._servicesQueryDef.Sort = args.GetQuerySort(nameof(Car.RegNumber));
        this._servicesQueryDef.FullText = args.Filter ?? string.Empty;

        this._servicesQueryDef.Page = args.GetQueryPage();
        this._servicesQueryDef.PageSize = args.GetQueryPageSize();

        try
        {
            var result = await queryService.GetListAsync<Car>(
                        this._servicesQueryDef);

            if (result != null)
            {
                this.Data = result.Items;
                this.RowCount = result.TotalRows;
            }
        }
        finally
        {
            this.isLoading = false;
            this.StateHasChanged();
        }
    }

    private async Task OnChange (object value)
    {
        var xx = grid.SelectedItem;

    }
